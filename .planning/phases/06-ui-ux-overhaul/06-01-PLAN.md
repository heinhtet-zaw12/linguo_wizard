IMPORTANT: The legacy aliases use `static Color get` (not `static const Color`) because they delegate to a runtime value. This means existing code like `AppColors.primaryPink` will compile but the values will now respect the active theme (light or dark).

However, there's a problem: existing code uses `const` in places like `const BoxDecoration(color: AppColors.bgTop)` or `const BoxDecoration(border: Border(left: BorderSide(color: AppColors.accentGold)))`. Since the getters are no longer `const`, any `const` context will fail to compile.

To handle this: Keep the old `static const Color` fields as well, but mark them as deprecated. The new getters take precedence. For places where `const` is required (like in `BoxDecoration`), the existing code will need to be updated in Wave 2/3 — but for now, we need to ensure compilation.

ACTUALLY — the safest approach is:
1. Keep the old `static const Color` fields EXACTLY as they are (unchanged)
2. Add NEW getters for the new token names that delegate to the provider
3. This way ALL existing code compiles unchanged

import 'package:flutter/material.dart';
import 'package:google_fonts/google_fonts.dart';

import 'app_colors.dart';
import 'app_text_styles.dart';

/// Application theme configuration.
/// Provides both light and dark Material 3 themes.
class AppTheme {
  AppTheme._();
  
  static ThemeData get light {
    final colorScheme = ColorScheme.fromSeed(
      seedColor: AppColors.accentPrimary,
      brightness: Brightness.light,
      primary: const LightColors().accentPrimary,
      onPrimary: const LightColors().textOnAccent,
      surface: const LightColors().surfacePrimary,
      onSurface: const LightColors().textPrimary,
    );
    
    return ThemeData(
      brightness: Brightness.light,
      useMaterial3: true,
      scaffoldBackgroundColor: const LightColors().backgroundStart,
      colorScheme: colorScheme,
      textTheme: _buildTextTheme(Brightness.light),
      appBarTheme: AppBarTheme(
        backgroundColor: Colors.transparent,
        elevation: 0,
        scrolledUnderElevation: 0,
        titleTextStyle: AppTextStyles.headingMedium(
          color: const LightColors().textPrimary,
        ),
      ),
      navigationBarTheme: NavigationBarThemeData(
        backgroundColor: const LightColors().surfaceGlass,
        elevation: 0,
        indicatorColor: const LightColors().accentPrimary,
        labelTextStyle: WidgetStateProperty.resolveWith((states) {
          if (states.contains(WidgetState.selected)) {
            return AppTextStyles.labelMedium(
              color: const LightColors().textOnAccent,
            );
          }
          return AppTextStyles.labelMedium(
            color: const LightColors().textTertiary,
          );
        }),
      ),
    );
  }
  
  static ThemeData get dark {
    final colorScheme = ColorScheme.fromSeed(
      seedColor: const DarkColors().accentPrimary,
      brightness: Brightness.dark,
      primary: const DarkColors().accentPrimary,
      onPrimary: const DarkColors().textOnAccent,
      surface: const DarkColors().surfacePrimary,
      onSurface: const DarkColors().textPrimary,
    );
    
    return ThemeData(
      brightness: Brightness.dark,
      useMaterial3: true,
      scaffoldBackgroundColor: const DarkColors().backgroundStart,
      colorScheme: colorScheme,
      textTheme: _buildTextTheme(Brightness.dark),
      appBarTheme: AppBarTheme(
        backgroundColor: Colors.transparent,
        elevation: 0,
        scrolledUnderElevation: 0,
        titleTextStyle: AppTextStyles.headingMedium(
          color: const DarkColors().textPrimary,
        ),
      ),
      navigationBarTheme: NavigationBarThemeData(
        backgroundColor: const DarkColors().surfaceGlass,
        elevation: 0,
        indicatorColor: const DarkColors().accentPrimary,
        labelTextStyle: WidgetStateProperty.resolveWith((states) {
          if (states.contains(WidgetState.selected)) {
            return AppTextStyles.labelMedium(
              color: const DarkColors().textOnAccent,
            );
          }
          return AppTextStyles.labelMedium(
            color: const DarkColors().textTertiary,
          );
        }),
      ),
    );
  }
  
  /// Build a TextTheme using GoogleFonts.
  static TextTheme _buildTextTheme(Brightness brightness) {
    final color = brightness == Brightness.light
        ? const LightColors().textPrimary
        : const DarkColors().textPrimary;
    
    final secondaryColor = brightness == Brightness.light
        ? const LightColors().textSecondary
        : const DarkColors().textSecondary;
    
    return TextTheme(
      displayLarge: AppTextStyles.displayLarge(color: color),
      displayMedium: AppTextStyles.displayMedium(color: color),
      headlineLarge: AppTextStyles.headingLarge(color: color),
      headlineMedium: AppTextStyles.headingMedium(color: color),
      headlineSmall: AppTextStyles.headingSmall(color: color),
      bodyLarge: AppTextStyles.bodyLarge(color: secondaryColor),
      bodyMedium: AppTextStyles.bodyMedium(color: secondaryColor),
      bodySmall: AppTextStyles.bodySmall(color: secondaryColor),
      labelLarge: AppTextStyles.labelLarge(color: color),
      labelMedium: AppTextStyles.labelMedium(color: secondaryColor),
      labelSmall: AppTextStyles.labelSmall(color: secondaryColor),
    );
  }
  
  /// Update the AppColorProvider based on brightness.
  /// Call this when theme changes (e.g., dark mode toggle).
  static void updateColorProvider(Brightness brightness) {
    if (brightness == Brightness.dark) {
      AppColorProvider.update(const DarkColors());
    } else {
      AppColorProvider.update(const LightColors());
    }
  }
}
